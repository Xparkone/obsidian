```
sudo ./remove-audit.sh zkyhistory                # 卸载 bash+fish 两个版本，保留已有日志
sudo ./remove-audit.sh zkyhistory --purge-logs   # 同时删除日志文件/目录
sudo ./remove-audit.sh zkyhistory --bash-only    # 只卸载 bash 版
sudo ./remove-audit.sh zkyhistory --fish-only    # 只卸载 fish 版
```


```
#!/usr/bin/env bash
#
# remove-audit.sh
#
# 卸载通过 setup-bash-audit.sh / setup-fish-audit.sh 部署的命令审计钩子。
#
# 用法:
#   sudo ./remove-audit.sh <前缀名> [选项]
#
# 选项:
#   --keep-logs   保留已产生的日志文件/目录（默认: 保留）
#   --purge-logs  连同日志文件/目录一起删除
#   --bash-only   只卸载 bash 版
#   --fish-only   只卸载 fish 版
#   (不加 --bash-only/--fish-only 时，两个版本都会尝试卸载)
#
# 示例:
#   sudo ./remove-audit.sh zkyhistory                # 卸载全部，保留日志
#   sudo ./remove-audit.sh zkyhistory --purge-logs   # 卸载全部，日志也删掉
#   sudo ./remove-audit.sh zkyhistory --fish-only    # 只卸载 fish 版

set -euo pipefail

# ---------- 参数处理 ----------
if [ $# -lt 1 ]; then
  echo "用法: sudo $0 <前缀名> [--keep-logs|--purge-logs] [--bash-only|--fish-only]" >&2
  exit 1
fi

NAME="$1"
shift || true

PURGE_LOGS=false
ONLY_BASH=false
ONLY_FISH=false

for arg in "$@"; do
  case "$arg" in
    --keep-logs)  PURGE_LOGS=false ;;
    --purge-logs) PURGE_LOGS=true ;;
    --bash-only)  ONLY_BASH=true ;;
    --fish-only)  ONLY_FISH=true ;;
    *)
      echo "错误: 未知参数 $arg" >&2
      exit 1
      ;;
  esac
done

if ! [[ "$NAME" =~ ^[A-Za-z][A-Za-z0-9_]*$ ]]; then
  echo "错误: 名称只能以字母开头，且只包含字母、数字、下划线。你传入的是: $NAME" >&2
  exit 1
fi

if [ "$EUID" -ne 0 ]; then
  echo "错误: 请用 root 权限运行此脚本（sudo）。" >&2
  exit 1
fi

DO_BASH=true
DO_FISH=true
if [ "$ONLY_BASH" = true ]; then DO_FISH=false; fi
if [ "$ONLY_FISH" = true ]; then DO_BASH=false; fi

echo "==> 卸载审计钩子，名称前缀: ${NAME}"
echo "    卸载 bash 版: ${DO_BASH}"
echo "    卸载 fish 版: ${DO_FISH}"
echo "    删除日志:     ${PURGE_LOGS}"
echo

REMOVED_ANY=false

# ---------- bash 版卸载 ----------
if [ "$DO_BASH" = true ]; then
  PROFILE_SCRIPT="/etc/profile.d/${NAME}.sh"
  RSYSLOG_CONF_BASH="/etc/rsyslog.d/${NAME}.conf"
  LOG_DIR_BASH="/var/log/${NAME}"
  LOG_FILE_BASH="${LOG_DIR_BASH}/history.log"

  echo "==> [bash] 清理 ${PROFILE_SCRIPT}"
  if [ -f "$PROFILE_SCRIPT" ]; then
    rm -f "$PROFILE_SCRIPT"
    echo "    已删除"
    REMOVED_ANY=true
  else
    echo "    不存在，跳过"
  fi

  echo "==> [bash] 清理 ${RSYSLOG_CONF_BASH}"
  if [ -f "$RSYSLOG_CONF_BASH" ]; then
    rm -f "$RSYSLOG_CONF_BASH"
    echo "    已删除"
    REMOVED_ANY=true
  else
    echo "    不存在，跳过"
  fi

  if [ "$PURGE_LOGS" = true ]; then
    if [ -d "$LOG_DIR_BASH" ]; then
      rm -rf "$LOG_DIR_BASH"
      echo "    已删除日志目录 ${LOG_DIR_BASH}"
    fi
  else
    if [ -f "$LOG_FILE_BASH" ]; then
      echo "    保留日志文件: ${LOG_FILE_BASH}（如需清除请加 --purge-logs 重新执行，或手动 rm -rf ${LOG_DIR_BASH}）"
    fi
  fi
  echo
fi

# ---------- fish 版卸载 ----------
if [ "$DO_FISH" = true ]; then
  FISH_SCRIPT="/etc/fish/conf.d/${NAME}.fish"
  RSYSLOG_CONF_FISH="/etc/rsyslog.d/${NAME}-fish.conf"
  LOG_DIR_FISH="/var/log/${NAME}"
  LOG_FILE_FISH="${LOG_DIR_FISH}/fish-history.log"

  echo "==> [fish] 清理 ${FISH_SCRIPT}"
  if [ -f "$FISH_SCRIPT" ]; then
    rm -f "$FISH_SCRIPT"
    echo "    已删除"
    REMOVED_ANY=true
  else
    echo "    不存在，跳过"
  fi

  echo "==> [fish] 清理 ${RSYSLOG_CONF_FISH}"
  if [ -f "$RSYSLOG_CONF_FISH" ]; then
    rm -f "$RSYSLOG_CONF_FISH"
    echo "    已删除"
    REMOVED_ANY=true
  else
    echo "    不存在，跳过"
  fi

  if [ "$PURGE_LOGS" = true ]; then
    if [ -f "$LOG_FILE_FISH" ]; then
      rm -f "$LOG_FILE_FISH"
      echo "    已删除日志文件 ${LOG_FILE_FISH}"
    fi
    # 如果目录下已经没有其他文件（比如 bash 版日志也一起删了或本来没有），顺手清空目录
    if [ -d "$LOG_DIR_FISH" ] && [ -z "$(ls -A "$LOG_DIR_FISH" 2>/dev/null)" ]; then
      rmdir "$LOG_DIR_FISH"
      echo "    日志目录已空，已删除 ${LOG_DIR_FISH}"
    fi
  else
    if [ -f "$LOG_FILE_FISH" ]; then
      echo "    保留日志文件: ${LOG_FILE_FISH}"
    fi
  fi
  echo
fi

# ---------- 重启 rsyslog 使配置移除生效 ----------
if [ "$REMOVED_ANY" = true ]; then
  echo "==> 重启 rsyslog 以应用配置变更"
  systemctl restart rsyslog
  if systemctl is-active --quiet rsyslog; then
    echo "    OK: rsyslog 运行正常"
  else
    echo "    警告: rsyslog 重启后状态异常，请检查 systemctl status rsyslog" >&2
  fi
else
  echo "==> 未找到任何相关文件，无需重启 rsyslog"
fi

echo
echo "==================================================================="
echo "卸载完成。"
echo
echo "重要提示: 已经打开的 bash / fish 会话里，钩子函数仍驻留在当前 shell"
echo "的内存/环境中（PROMPT_COMMAND 或 fish 的 --on-event 绑定不会自动清除），"
echo "如需彻底停止记录，请让用户重新打开一个新的终端/SSH 会话，"
echo "或在当前会话手动执行以下操作："
echo
echo "  bash 会话内手动解除:"
echo "    unset -f __${NAME}_audit_bash 2>/dev/null || true"
echo "    PROMPT_COMMAND=\$(echo \"\$PROMPT_COMMAND\" | sed \"s/__${NAME}_audit_bash;\\?//\")"
echo
echo "  fish 会话内手动解除:"
echo "    functions -e __${NAME}_audit_fish"
echo "==================================================================="
```
