#!/usr/bin/env bash
#
# setup-audit-logrotate.sh
#
# 为 setup-bash-audit.sh / setup-fish-audit.sh 产生的审计日志配置 logrotate。
#
# 用法:
#   sudo ./setup-audit-logrotate.sh <前缀名> [选项]
#
# 选项:
#   --days N        保留 N 天的轮转日志（默认 30）
#   --size SIZE     达到该大小就轮转，如 100M（默认不按大小，只按时间，daily）
#   --compress      压缩轮转后的旧日志（默认开启）
#   --no-compress   不压缩
#
# 示例:
#   sudo ./setup-audit-logrotate.sh zkyhistory
#   sudo ./setup-audit-logrotate.sh zkyhistory --days 90
#   sudo ./setup-audit-logrotate.sh zkyhistory --size 50M --days 60

set -euo pipefail

# ---------- 参数处理 ----------
if [ $# -lt 1 ]; then
  echo "用法: sudo $0 <前缀名> [--days N] [--size SIZE] [--compress|--no-compress]" >&2
  exit 1
fi

NAME="$1"
shift || true

DAYS=30
SIZE=""
COMPRESS=true

while [ $# -gt 0 ]; do
  case "$1" in
    --days)
      DAYS="$2"
      shift 2
      ;;
    --size)
      SIZE="$2"
      shift 2
      ;;
    --compress)
      COMPRESS=true
      shift
      ;;
    --no-compress)
      COMPRESS=false
      shift
      ;;
    *)
      echo "错误: 未知参数 $1" >&2
      exit 1
      ;;
  esac
done

if ! [[ "$NAME" =~ ^[A-Za-z][A-Za-z0-9_]*$ ]]; then
  echo "错误: 名称只能以字母开头，且只包含字母、数字、下划线。你传入的是: $NAME" >&2
  exit 1
fi

if ! [[ "$DAYS" =~ ^[0-9]+$ ]]; then
  echo "错误: --days 必须是正整数，你传入的是: $DAYS" >&2
  exit 1
fi

if [ "$EUID" -ne 0 ]; then
  echo "错误: 请用 root 权限运行此脚本（sudo）。" >&2
  exit 1
fi

LOG_DIR="/var/log/${NAME}"
LOG_FILE_BASH="${LOG_DIR}/history.log"
LOG_FILE_FISH="${LOG_DIR}/fish-history.log"
LOGROTATE_CONF="/etc/logrotate.d/${NAME}"

echo "==> 配置 logrotate，名称前缀: ${NAME}"
echo "    配置文件: ${LOGROTATE_CONF}"
echo "    保留天数: ${DAYS}"
[ -n "$SIZE" ] && echo "    轮转阈值: ${SIZE}"
echo "    压缩:     ${COMPRESS}"
echo

# ---------- 检查日志文件是否存在 ----------
FOUND_ANY=false
for f in "$LOG_FILE_BASH" "$LOG_FILE_FISH"; do
  if [ -f "$f" ]; then
    echo "    发现日志文件: $f"
    FOUND_ANY=true
  fi
done
if [ "$FOUND_ANY" = false ]; then
  echo "警告: 未发现 ${LOG_DIR} 下的任何日志文件，仍会写入 logrotate 配置" >&2
  echo "      （logrotate 对不存在的文件会跳过，不会报错）" >&2
fi
echo

# ---------- 生成 logrotate 配置 ----------
echo "==> 生成 ${LOGROTATE_CONF}"

COMPRESS_LINES=""
if [ "$COMPRESS" = true ]; then
  COMPRESS_LINES="    compress
    delaycompress"
else
  COMPRESS_LINES="    nocompress"
fi

SIZE_LINE=""
if [ -n "$SIZE" ]; then
  SIZE_LINE="    size ${SIZE}"
fi

cat >"${LOGROTATE_CONF}" <<EOF
${LOG_FILE_BASH}
${LOG_FILE_FISH}
{
    daily
${SIZE_LINE}
    rotate ${DAYS}
    missingok
    notifempty
${COMPRESS_LINES}
    dateext
    dateformat -%Y%m%d
    create 0640 syslog adm
    sharedscripts
    postrotate
        systemctl kill -s HUP rsyslog.service >/dev/null 2>&1 || true
    endscript
}
EOF

# 去掉空行（比如没传 --size 时的空行）
sed -i '/^$/d' "${LOGROTATE_CONF}"

echo "    OK"
echo

# ---------- 校验配置语法 ----------
echo "==> 校验 logrotate 配置"

if command -v logrotate >/dev/null 2>&1; then
  logrotate -d "${LOGROTATE_CONF}" >/tmp/logrotate-debug.$$.log 2>&1 || {
    echo "错误: logrotate 配置语法有问题，输出如下：" >&2
    cat /tmp/logrotate-debug.$$.log >&2
    rm -f /tmp/logrotate-debug.$$.log
    exit 1
  }
  rm -f /tmp/logrotate-debug.$$.log
  echo "    OK: 语法检查通过（-d 为 dry-run，不会真正轮转）"
else
  echo "    警告: 未找到 logrotate 命令，跳过语法检查" >&2
fi
echo

echo "==================================================================="
echo "配置完成。"
echo
echo "logrotate 通常由系统的每日 cron/timer 自动触发，一般在："
echo "    /etc/cron.daily/logrotate"
echo "或者较新系统上通过 systemd timer："
echo "    systemctl status logrotate.timer"
echo
echo "手动测试立即触发一次轮转（会真的轮转，谨慎在生产环境使用）:"
echo "    sudo logrotate -f ${LOGROTATE_CONF}"
echo
echo "只做 dry-run 不实际轮转:"
echo "    sudo logrotate -d ${LOGROTATE_CONF}"
echo
echo "查看当前配置:"
echo "    cat ${LOGROTATE_CONF}"
echo "==================================================================="
